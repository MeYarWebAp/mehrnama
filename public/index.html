async function handleLogin() { let identifier = document.getElementById('loginPhoneOrId').value.trim(); let password = document.getElementById('loginPass').value.trim(); if (!identifier || !password) { alert('لطفاً اطلاعات ورود و رمز عبور را کامل وارد کنید.'); return; } try { let res = await fetch('/api/login', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ phone_or_id: identifier, password: password, user_type: currentLoginType }) }); let data = await res.json(); if (data.success) { localStorage.setItem('mehrnama_user', JSON.stringify(data.user)); closeLoginModal(); updateHeaderUser(); alert('ورود با موفقیت انجام شد!'); location.reload(); } else { alert(data.error || 'خطا در ورود به سامانه.'); } } catch (err) { console.error(err); alert('خطا در ارتباط با سرور.'); } } async function handleRegister() { let isReal = currentRegType === 'real'; let name = isReal ? document.getElementById('regRealName').value.trim() : document.getElementById('regLegalName').value.trim(); let phone_or_id = isReal ? document.getElementById('regRealPhone').value.trim() : document.getElementById('regLegalRegNo').value.trim(); let password = isReal ? document.getElementById('regRealPass').value.trim() : document.getElementById('regLegalPass').value.trim(); if (!name || !phone_or_id || !password) { alert('لطفاً تمامی فیلدهای الزامی را پر کنید.'); return; } let payload = { name: name, user_type: currentRegType, phone_or_id: phone_or_id, password: password, province: !isReal ? document.getElementById('regLegalProvince').value.trim() : null, city: !isReal ? document.getElementById('regLegalCity').value.trim() : null, district: !isReal ? document.getElementById('regLegalDistrict').value.trim() : null }; try { let res = await fetch('/api/register', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); let data = await res.json(); if (data.success) { localStorage.setItem('mehrnama_user', JSON.stringify({ id: data.user.id, name: data.user.name, type: data.user.user_type })); closeRegisterModal(); updateHeaderUser(); alert('ثبت‌نام و ورود با موفقیت در دیتابیس انجام شد!'); location.reload(); } else { alert(data.error || 'خطا در ثبت‌نام.'); } } catch (err) { console.error(err); alert('خطا در ارتباط با سرور.'); } }
